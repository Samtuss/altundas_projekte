create table if not exists r_raume (
r_id varchar(5) primary key,
r_platze varchar(50)
);

create table if not exists g_gegenstande (
g_id varchar(5) primary key,
g_bez varchar(50)
);

create table if not exists gs_geschlechter(
gs_id int(11) primary key,
gs_mw char(1),
gs_mannlichweiblich varchar(10),
gs_schuler varchar(15),
gs_lehrer varchar(15)
);

create table if not exists s_schueler (
s_nr integer primary key unique,
s_name varchar(35) not null,
s_vorname varchar(35) not null,
s_gs_geschlecht int(11),
s_gebdat date null,
s_plz int(11),
s_ort varchar(35),
s_strasse varchar(50),
s_k_klasse char(6) null
);

create table if not exists k_klassen (
k_id varchar(6) primary key,
k_bez varchar(35) not null,
k_s_klaspr integer null unique,
k_s_klasprstv integer null unique,
k_l_klavst varchar(5) null,
k_a_abteilung int,
foreign key (k_s_klasprstv)
	references s_schueler (s_nr)
    on delete set null on update cascade,
foreign key (k_s_klaspr)
	references s_schueler (s_nr)
    on delete set null on update cascade
);

create table if not exists a_abteilungen(
a_id int(11) primary key,
a_kurzel varchar(5),
a_abtname varchar(70),
a_l_abtleiter varchar(5)
);

create table if not exists l_lehrer(
l_id varchar(5) primary key,
l_vorname varchar(30) not null,
l_nachname varchar(35) not null,
l_gebdat date null,
l_sprechstunde varchar(25) null,
l_gehalt decimal(6,2) not null,
l_eintrittsjahr int(11) not null,
l_gs_geschlecht int(11) not null,
l_a_abteilung int(11) not null,
l_l_chef varchar(5),

foreign key (l_gs_geschlecht)
	references gs_geschlechter (gs_id),
    
foreign key (l_a_abteilung)
	references a_abteilungen (a_id),
    
foreign key (l_l_chef)
	references l_lehrer (l_id)
);

create table if not exists st_stunden(
st_k_klasse varchar(6),
st_g_fach varchar(5),
st_stunde varchar(5) primary key,
st_l_lehrer varchar(5),
st_r_raum varchar(5),

foreign key (st_r_raum)
	references r_raume (r_id),
    
foreign key (st_l_lehrer)
	references l_lehrer (l_id),

foreign key (st_g_fach)
	references g_gegenstande (g_id),
    
foreign key (st_k_klasse)
	references k_klassen (k_id)

);

create table if not exists p_pruefungen(
p_datum date primary key,
p_s_kandidat int(11),
p_l_prufer varchar(5),
p_g_fach varchar(5),
p_art varchar(3),
p_note int(11),

foreign key (p_s_kandidat)
	references s_schueler (s_nr),
    
foreign key (p_l_prufer)
	references l_lehrer (l_id),
    
foreign key (p_g_fach)
	references g_gegenstande (g_id)
);

alter table a_abteilungen add foreign key(a_l_abtleiter)
